5. Implement heartbeat audio that plays when "Tap to Listen" is visible (when not actively speaking to Clara) to enhance the living person experience
6. UI heartbeat visual and audio changes feel natural and enhance the sense of talking to a real person without interfering with existing conversation flow

**Heart Icon Heartbeat Specification**:
```typescript
// SIMPLE HEART ICON SPECIFICATION - Based on UX Research
interface HeartbeatIconSpec {
  position: 'top-right';       // Screen position: 16px from top and right edges
  size: 24;                    // px - Small, aesthetically pleasing size
  icon: 'Heart';               // Lucide React Heart icon
  animationType: 'double-pulse'; // Realistic heartbeat pattern
  scaleSequence: [1.0, 1.2, 1.0, 1.1, 1.0]; // Double-beat pattern
  dropShadow: '0 2px 4px rgba(0,0,0,0.2)'; // Subtle depth
  zIndex: 50;                  // Always visible above other elements
}

// BPM TO CYCLE TIME MAPPING - Scientifically Based
const heartbeatBpmConfigs: Record<EmotionalMood, { bpm: number; cycleMs: number; color: string }> = {
  neutral: { bpm: 60, cycleMs: 1000, color: '#9ca3af' },     // Gray
  excited: { bpm: 90, cycleMs: 667, color: '#ec4899' },      // Pink
  calm: { bpm: 45, cycleMs: 1333, color: '#10b981' },        // Green
  angry: { bpm: 100, cycleMs: 600, color: '#ef4444' },       // Red
  happy: { bpm: 75, cycleMs: 800, color: '#f59e0b' },        // Golden
  sad: { bpm: 50, cycleMs: 1200, color: '#3b82f6' },         // Blue

}

// CONVERSATION INTENSITY EFFECTS - Data Driven
interface ConversationIntensityEffects {
  low: { scaleBonus: 0, glowIntensity: 0, bpmMultiplier: 1.0 };
  medium: { scaleBonus: 0, glowIntensity: 0, bpmMultiplier: 1.1 };
  high: { scaleBonus: 0.1, glowIntensity: 0.3, bpmMultiplier: 1.3 }; // Stronger pulse + glow
}
```

**Exact React Component Structure**:
```typescript
// HeartbeatIcon.tsx - Simple heart icon with realistic pulse animation
import { Heart } from 'lucide-react';
import { motion } from 'framer-motion';

interface HeartbeatIconProps {
  mood: EmotionalMood;
  conversationIntensity: 'low' | 'medium' | 'high';
}

const HeartbeatIcon = ({ mood, conversationIntensity }: HeartbeatIconProps) => {
  const config = heartbeatBpmConfigs[mood];
  const effects = conversationIntensityEffects[conversationIntensity];

  return (
    <motion.div
      className="fixed top-4 right-4 z-50"
      style={{
        filter: `drop-shadow(0 2px 4px rgba(0,0,0,0.2))${
          effects.glowIntensity > 0 ? ` drop-shadow(0 0 8px ${config.color}${Math.round(effects.glowIntensity * 255).toString(16)})` : ''
        }`
      }}
      animate={{
        scale: [
          1.0,
          1.2 + effects.scaleBonus,
          1.0,
          1.1 + effects.scaleBonus * 0.5,
          1.0
        ]
      }}
      transition={{
        duration: config.cycleMs / 1000 / effects.bpmMultiplier,
        repeat: Infinity,
        ease: 'easeOut',
        times: [0, 0.2, 0.3, 0.5, 1.0] // Double-pulse timing
      }}
    >
      <Heart
        size={24}
        color={config.color}
        fill={config.color}
        style={{
          transition: 'color 0.3s ease-in-out' // Smooth color transitions
        }}
      />
    </motion.div>
  );
};

// Conversation Page Integration - add to conversation/page.tsx
{/* Add HeartbeatIcon component */}
<HeartbeatIcon
  mood={currentMood}
  conversationIntensity={conversationIntensity}
/>
```

- Heartbeat visual rendering: <16ms per frame for 60fps
